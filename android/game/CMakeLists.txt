# libreeot_game.so - the recompiled game as a loadable Android shared library.
#
# Upstream builds the game as an executable (`reeot`), which cannot be loaded
# in-process on Android (SDL's Android surface glue lives in the app process).
# This project builds the SAME game sources as a SHARED library that exports
# `reeot_windowed_app_creator` (see patches/upstream/0003), which the launcher
# (libedgeoftime.so) resolves with SDL_LoadFunction at its GAME BOOTSTRAP HOOK.
#
# It includes upstream's whole CMake project so that everything upstream's
# build produces is reused: the pkz packages + embedded asset sources, the
# automatic `rexglue codegen` run (reeot_codegen) when the game files are
# present, the reeot_GameLogic guest DLL, and the reeot_shader_cache object
# library. Nothing is duplicated.
#
# Required CMake arguments:
#   REXGLUE_SDK_DIR    - patched rexglue-sdk source tree
#   UPSTREAM_DIR       - the upstream (goliathret/EdgeOfTimeRecomp) checkout,
#                        with patches/upstream/ applied
#   REXGLUE_BUILD_DIR  - SDK build output dir (unused directly; the SDK is built
#                        as part of add_subdirectory below, but kept for the
#                        CI scripts' symmetry)
#   REX_ANDROID_SUPPORT_INCLUDE - android/rex_android/include
#
# Without codegen output (no game files), main.cpp/reeot_app.cpp are excluded
# (they include generated headers) and the library builds without the creator
# export - the launcher then falls back to its status screen. This keeps the
# configure+build working in CI without any game files.

cmake_minimum_required(VERSION 3.22)
project(reeot_game_android LANGUAGES CXX)

if(NOT REXGLUE_SDK_DIR OR NOT UPSTREAM_DIR)
    message(FATAL_ERROR "REXGLUE_SDK_DIR and UPSTREAM_DIR are required")
endif()

# Upstream's full project: pkz packages, embedded assets, codegen wiring,
# rex::runtime / rex::rexglue targets, rexglue_setup_target() macro.
add_subdirectory("${UPSTREAM_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/upstream")

# --- Game sources (mirror upstream's REEOT_SOURCES via glob) ---------------
file(GLOB_RECURSE REEOT_GAME_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/src/*.cpp")
list(REMOVE_ITEM REEOT_GAME_SOURCES "${UPSTREAM_DIR}/src/platform/moltenvk.cpp")

set(REEOT_GENERATED_SOURCES "")
set(REEOT_HAS_CODEGEN FALSE)
if(EXISTS "${UPSTREAM_DIR}/generated/default/sources.cmake")
    include("${UPSTREAM_DIR}/generated/default/sources.cmake")
    set(REEOT_GENERATED_SOURCES ${GENERATED_SOURCES})
    set(REEOT_HAS_CODEGEN TRUE)
    message(STATUS "reeot_game: codegen output found (${REEOT_HAS_CODEGEN})")
else()
    message(STATUS "reeot_game: no codegen output - building without main.cpp/reeot_app.cpp")
    list(REMOVE_ITEM REEOT_GAME_SOURCES
        "${UPSTREAM_DIR}/src/main.cpp"
        "${UPSTREAM_DIR}/src/reeot_app.cpp")
endif()

add_library(reeot_game SHARED ${REEOT_GAME_SOURCES} ${REEOT_GENERATED_SOURCES})

# Embedded assets generated by upstream's CMake into its binary dir
# (embed/*.cpp: res/embed files + the pkz packages).
file(GLOB REEOT_EMBED_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/upstream/generated/embed/*.cpp")
target_sources(reeot_game PRIVATE ${REEOT_EMBED_SOURCES})

# The shader cache object library (real cache or the empty stub).
target_link_libraries(reeot_game PRIVATE $<TARGET_OBJECTS:reeot_shader_cache>)

target_include_directories(reeot_game PRIVATE
    "${UPSTREAM_DIR}/src"
    "${CMAKE_CURRENT_BINARY_DIR}/upstream/generated"
    "${UPSTREAM_DIR}/generated/default"
    "${UPSTREAM_DIR}/thirdparty/stb"
    "${UPSTREAM_DIR}/thirdparty/XenosRecomp/XenosRecomp"
    "${UPSTREAM_DIR}/thirdparty/XenosRecomp/thirdparty/smol-v/source"
    "${UPSTREAM_DIR}/thirdparty/renderdoc"
    "${UPSTREAM_DIR}/thirdparty/plume"
    "${UPSTREAM_DIR}/thirdparty/plume/contrib/volk"
    "${UPSTREAM_DIR}/thirdparty/XenosRecomp/thirdparty/zstd/lib"
    "${REXGLUE_SDK_DIR}/thirdparty/imgui"
    "${REXGLUE_SDK_DIR}/thirdparty/tomlplusplus/include"
    "${REXGLUE_SDK_DIR}/thirdparty/vulkan-memory-allocator/include"
    "${REXGLUE_SDK_DIR}/thirdparty/xxHash"
    "${REX_ANDROID_SUPPORT_INCLUDE}"
)

target_compile_definitions(reeot_game PRIVATE
    _ITERATOR_DEBUG_LEVEL=0
    REEOT_BUILD_INSTALLER
    NOMINMAX
    WIN32_LEAN_AND_MEAN
)

# Same flag set as the SDK/upstream Android builds (ABI must match).
target_compile_options(reeot_game PRIVATE
    -fexperimental-library
    -ffp-model=strict
    -fPIC
    -O3
    -DNDEBUG
)

# SDK wiring: links rex::runtime, applies the SDK target settings, and with
# GPU_PLUGINS xenos links the XenosRecomp shader-translation plugin.
rexglue_setup_target(reeot_game GPU_PLUGINS xenos)

# 16 KB page alignment for Android 15+ devices.
target_link_options(reeot_game PRIVATE
    -Wl,-z,max-page-size=16384
    -Wl,-z,common-page-size=16384
)

# The guest GameLogic DLL (reeot_GameLogic) is defined by the codegen-emitted
# dll_targets.cmake (included by upstream's CMakeLists above) and builds as
# part of the default target set; it is packaged into the APK alongside this
# library by CI.
